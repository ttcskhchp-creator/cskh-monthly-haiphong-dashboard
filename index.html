// ================== NHIỆM VỤ 6 TIẾNG AN TOÀN (KHẮC PHỤC KẸT BỐC THĂM) ================== const SIX_HOURS = 6 * 60 * 60 * 1000; function getActiveQuest() { try { let questData = localStorage.getItem('magicCurrentQuest'); if (questData) { let q = JSON.parse(questData); if (Date.now() - q.createdAt < SIX_HOURS) { return q; } } } catch (e) { console.warn('Lỗi đọc quest cũ, tạo mới:', e); } return generateNew6HourQuest(); } function generateNew6HourQuest() { // Giá trị mặc định an toàn nếu chưa load kịp bé từ Cloud let rankScore = 150; if (typeof cachedKids !== 'undefined' && cachedKids.length) { const kid = cachedKids.find(k => k.id === currentKidId); const rank = getKidRank((kid && kid.weekly_done) || 0); if (rank && rank.minScore) rankScore = rank.minScore; } const targetStories = Math.random() > 0.5 ? 2 : 1; const targetGames = Math.floor(Math.random() * 2 + 2); // 2 - 3 lượt chơi const targetSentences = Math.floor(Math.random() * 2 + 1); // 1 - 2 câu đọc const targetScore = rankScore; const newQuest = { id: 'quest_' + Date.now(), createdAt: Date.now(), targetStories: targetStories, targetGames: targetGames, targetSentences: targetSentences, targetScore: targetScore, attempts: {}, completedKids: [] }; try { localStorage.setItem('magicCurrentQuest', JSON.stringify(newQuest)); } catch (e) {} sessionTracker = { readStories: new Set(), sentencesPassed: 0, gamesPlayed: 0, scoreEarned: 0 }; return newQuest; } function renderQuestUI() { const content = document.getElementById('questContent'); if (!content) return; const q = getActiveQuest(); const now = Date.now(); const timeLeft = Math.max(0, SIX_HOURS - (now - (q.createdAt || now))); const hours = Math.floor(timeLeft / (60 * 60 * 1000)); const minutes = Math.floor((timeLeft % (60 * 60 * 1000)) / (60 * 1000)); const timerEl = document.getElementById('questTimer'); if (timerEl) { timerEl.textContent = `⏱️ ${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}:00`; } const attemptsUsed = (q.attempts && q.attempts[currentKidId]) || 0; const retryBadge = document.getElementById('retryBadge'); const retryBtn = document.getElementById('retryQuestBtn'); if (retryBadge) { if (attemptsUsed >= 3) { retryBadge.textContent = 'Đã hết 3 lượt thử 🛑'; } else { retryBadge.textContent = `Lượt thử: ${Math.min(3, attemptsUsed + 1)}/3`; } } if (retryBtn) { if (attemptsUsed >= 3) { retryBtn.classList.add('hidden'); } else { retryBtn.classList.remove('hidden'); } } const isCompleted = (q.completedKids || []).includes(currentKidId); // Render nội dung nhiệm vụ trực tiếp content.innerHTML = `
📚 1. Đọc và nghe: ${q.targetStories} bài truyện khác nhau
🎙️ 2. Luyện phát âm đúng: ${q.targetSentences} câu văn trong bài
🕹️ 3. Tham gia đố vui hoặc game: ${q.targetGames} lượt chơi
⭐ 4. Điểm đạt trong lượt này: ≥ ${q.targetScore} điểm
`; updateSessionUI(); const badge = document.getElementById('questStatusBadge'); const claimBtn = document.getElementById('claimQuestBtn'); if (badge && claimBtn) { if (isCompleted) { badge.textContent = 'Trạng thái: Đã hoàn thành xuất sắc! 🏆'; badge.className = 'text-xs font-black text-emerald-600'; claimBtn.disabled = true; claimBtn.textContent = '✅ Đã Ghi Điểm'; } else { badge.textContent = 'Trạng thái: Đang theo dõi các bước... ⏳'; badge.className = 'text-xs font-black text-orange-600'; claimBtn.disabled = false; claimBtn.textContent = '🏆 Nộp Ghi Điểm'; } } }
